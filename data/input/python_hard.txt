def limit_calls(max_calls):
    def decorator(func):
        calls = 0
        def wrapper(*args, **kwargs):
            nonlocal calls
            if calls >= max_calls:
                raise RuntimeError('Function call limit exceeded')
            calls += 1
            return func(*args, **kwargs)
        return wrapper
    return decorator

def levenshtein(s1, s2):
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]

    triples = []
    for a in range(1, limit+1):
        for b in range(a, limit+1):
            c = (a**2 + b**2) ** 0.5
            if c.is_integer() and c <= limit:

    n = len(vertices)
    area = 0
    for i in range(n):
        x1, y1 = vertices[i]
        x2, y2 = vertices[(i+1) % n]
        area += x1 * y2 - x2 * y1

    if n == 0:
        return '0'
    bits = []
    while n:
        bits.append(str(n & 1))
        n >>= 1
    return ''.join(reversed(bits))

def eval_postfix(expr):
    stack = []
    for token in expr.split():
        if token in '+-*/':
            b = stack.pop()
            a = stack.pop()
            if token == '+': stack.append(a + b)
            elif token == '-': stack.append(a - b)
            elif token == '*': stack.append(a * b)
            else: stack.append(a / b)
        else:
            stack.append(float(token))
    return stack[0]

def _transfer(self):
        if not self.out_stack:
            while self.in_stack:
                val, mn, mx = self.in_stack.pop()
                if self.out_stack:
                    new_min = min(val, self.out_stack[-1][1])
                    new_max = max(val, self.out_stack[-1][2])

                    new_min = new_max = val
                self.out_stack.append((val, new_min, new_max))

        self._transfer()
        if not self.out_stack:

        return self.out_stack.pop()[0]

        self._transfer()
        if not self.out_stack:
            raise IndexError('empty')

    n = len(sorted_nums)

def is_perfect(n):
    if n < 2:
        return False
    div_sum = 1
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            div_sum += i
            if i != n // i:
                div_sum += n // i
    return div_sum == n

def square_free(n):
    result = []
    i = 1
    while len(result) < n:
        if all(i % (j*j) != 0 for j in range(2, int(i**0.5)+1)):
