// === Structs ===
    struct Point
    {
        public int X { get; set; }
        public int Y { get; set; }
        public Point(int x, int y) { X = x; Y = y; }
        public override string ToString() => $"({X}, {Y})";
        public static Point operator +(Point a, Point b) => new Point(a.X + b.X, a.Y + b.Y);
    }

// === Base Class ===
    class Animal : IAnimal
    {
        public string Name { get; protected set; }
        public Animal(string name) { Name = name; }
        public virtual string Speak() => $"{Name} makes a noise.";
    }

// === Generic Class ===
    class Box<T>
    {
        public T Value { get; set; }
        public Box(T value) => Value = value;
        public override string ToString() => Value?.ToString() ?? "null";
    }

    class DataFetcher
    {
        public async Task<string> FetchDataAsync(string url)
        {
            // Simulate async HTTP call

            return $"Data from {url}";
        }
    }

// Palindrome
            static bool IsPalindrome(string str)
            {
                string cleaned = Regex.Replace(str.ToLower(), "[^a-z0-9]", "");
                return cleaned == new string(cleaned.Reverse().ToArray());
            }
            Console.WriteLine(IsPalindrome("Racecar")); // true
