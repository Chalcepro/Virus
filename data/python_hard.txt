### Response:
def limit_calls(max_calls):
    def decorator(func):
        calls = 0
        def wrapper(*args, **kwargs):
            nonlocal calls
            if calls >= max_calls:
                raise RuntimeError('Function call limit exceeded')
            calls += 1
            return func(*args, **kwargs)
        return wrapper
    return decorator

### Instruction:
Write a function that calculates the Levenshtein distance between two strings.

### Response:
def levenshtein(s1, s2):
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]

Write a class that implements a circular buffer with a fixed size. Oldest elements are overwritten when full.

### Response:
class CircularBuffer:
    def __init__(self, size):
        self.size = size
        self.buffer = [None] * size
        self.head = 0
        self.tail = 0
        self.full = False

    def write(self, item):
        self.buffer[self.tail] = item
        self.tail = (self.tail + 1) % self.size
        if self.full:
            self.head = (self.head + 1) % self.size
        self.full = self.tail == self.head

    def read_all(self):
        if not self.full and self.head == self.tail:
            return []
        if self.full:
            return self.buffer[self.head:] + self.buffer[:self.tail]
        return self.buffer[self.head:self.tail]

    n = len(vertices)
    area = 0
    for i in range(n):
        x1, y1 = vertices[i]

def eval_postfix(expr):
    stack = []
    for token in expr.split():
        if token in '+-*/':
            b = stack.pop()
            a = stack.pop()
            if token == '+': stack.append(a + b)
            elif token == '-': stack.append(a - b)
            elif token == '*': stack.append(a * b)
            else: stack.append(a / b)
        else:
            stack.append(float(token))
    return stack[0]

from functools import wraps
import time

def ttl_cache(ttl_seconds):
    def decorator(func):
        cache = {}
        @wraps(func)
        def wrapper(*args):
            now = time.time()
            if args in cache:
                value, timestamp = cache[args]
                if now - timestamp < ttl_seconds:
                    return value
            result = func(*args)
            cache[args] = (result, now)
            return result
        return wrapper
    return decorator

### Instruction:
Write a class `MinMaxQueue` that supports `push`, `pop`, `min`, `max` in O(1) average time (using two stacks).

### Response:
class MinMaxQueue:
    def __init__(self):
        self.in_stack = []
        self.out_stack = []

    def _transfer(self):
        if not self.out_stack:
            while self.in_stack:
                val, mn, mx = self.in_stack.pop()
                if self.out_stack:
                    new_min = min(val, self.out_stack[-1][1])
                    new_max = max(val, self.out_stack[-1][2])
                else:
                    new_min = new_max = val
                self.out_stack.append((val, new_min, new_max))

    def push(self, val):
        if self.in_stack:
            mn = min(val, self.in_stack[-1][1])
            mx = max(val, self.in_stack[-1][2])
        else:
            mn = mx = val
        self.in_stack.append((val, mn, mx))

    def pop(self):
        self._transfer()
        if not self.out_stack:
            raise IndexError('pop from empty queue')
        return self.out_stack.pop()[0]

    def min(self):
        self._transfer()
        if not self.out_stack:
            raise IndexError('empty')
        return self.out_stack[-1][1]

    def max(self):
        self._transfer()
        if not self.out_stack:
            raise IndexError('empty')
        return self.out_stack[-1][2]

def longest_palindrome(s):
    if not s:
        return ''
    start, max_len = 0, 1
    for i in range(len(s)):
        for l, r in [(i, i), (i, i+1)]:
            while l >= 0 and r < len(s) and s[l] == s[r]:
                if r - l + 1 > max_len:
                    start = l
                    max_len = r - l + 1
                l -= 1
                r += 1
    return s[start:start+max_len]

class Node:
    def __init__(self, value, next=None):
        self.value = value
        self.next = next

class LinkedList:
    def __init__(self):
        self.head = None

    def prepend(self, value):
        self.head = Node(value, self.head)

    def append(self, value):
        if not self.head:
            self.head = Node(value)
            return
        cur = self.head
        while cur.next:

def last_line(filename):

### Instruction:
Write a function that checks if a number is a perfect number (sum of proper divisors equals number).

### Response:
def is_perfect(n):
    if n < 2:
        return False
    div_sum = 1
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            div_sum += i
            if i != n // i:
                div_sum += n // i
    return div_sum == n

### Instruction:
Write a function that generates the first n square-free numbers (numbers not divisible by any perfect square other than 1).

### Response:
def square_free(n):
    result = []
    i = 1
    while len(result) < n:
        if all(i % (j*j) != 0 for j in range(2, int(i**0.5)+1)):
            result.append(i)
        i += 1
    return result

### Response:
class Polynomial:
    def __init__(self, coefficients):
        self.coeffs = coefficients
    def __call__(self, x):
        return sum(c * x**i for i, c in enumerate(self.coeffs))
