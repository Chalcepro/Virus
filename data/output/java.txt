// === Interfaces ===
interface Animal {
    String getName();
    String speak();
    default String info() { return "An animal."; }
}

// === Inheritance ===
class Dog extends Mammal {
    private String breed;
    public Dog(String name, String breed) {
        super(name);
        this.breed = breed;
    }
    @Override
    public String speak() { return name + " barks!"; }
    public String getBreed() { return breed; }
    public static String info() { return "Dogs are loyal."; }
}
