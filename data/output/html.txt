<script>
                class UserCard extends HTMLElement {
                    constructor() {
                        super();
                        const template = document.getElementById('user-card-template').content.cloneNode(true);
                        this.attachShadow({ mode: 'open' }).appendChild(template);
                    }
                    connectedCallback() {
                        this.shadowRoot.querySelector('img').src = this.getAttribute('avatar') || '';
                        this.shadowRoot.querySelector('strong').textContent = this.getAttribute('name') || '';
                        this.shadowRoot.querySelector('small').textContent = this.getAttribute('email') || '';
                    }
                }
                customElements.define('user-card', UserCard);
            </script>
