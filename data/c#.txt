using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Text.RegularExpressions;

namespace TrainingData
{
    // === Enums ===
    enum Status { Active, Inactive, Pending }

    // === Structs ===
    struct Point
    {
        public int X { get; set; }
        public int Y { get; set; }
        public Point(int x, int y) { X = x; Y = y; }
        public override string ToString() => $"({X}, {Y})";
        public static Point operator +(Point a, Point b) => new Point(a.X + b.X, a.Y + b.Y);
    }

    // === Interfaces ===
    interface IAnimal
    {
        string Name { get; }
        string Speak();
    }

    // === Base Class ===
    class Animal : IAnimal
    {
        public string Name { get; protected set; }
        public Animal(string name) { Name = name; }
        public virtual string Speak() => $"{Name} makes a noise.";
    }

    // === Derived Class ===
    class Dog : Animal
    {
        public string Breed { get; }
        public Dog(string name, string breed) : base(name)
        {
            Breed = breed;
        }
        public override string Speak() => $"{Name} barks!";
        public static string Info() => "Dogs are loyal.";
    }

    // === Generic Class ===
    class Box<T>
    {
        public T Value { get; set; }
        public Box(T value) => Value = value;
        public override string ToString() => Value?.ToString() ?? "null";
    }

    // === Delegates and Events ===
    delegate void Notify(string message);
    class Publisher
    {
        public event Notify OnNotify;
        public void RaiseEvent(string msg) => OnNotify?.Invoke(msg);
    }

    // === Async / Await Helpers ===
    class DataFetcher
    {
        public async Task<string> FetchDataAsync(string url)
        {
            // Simulate async HTTP call
            await Task.Delay(1000);
            return $"Data from {url}";
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== C# Training Data ===");

            // --- Variables and Primitive Types ---
            int age = 30;
            double price = 19.99;
            char grade = 'A';
            bool isActive = true;
            const float Pi = 3.14159f;
            string name = "Alice";
            
            // String interpolation
            Console.WriteLine($"{name} is {age} years old.");
            
            // --- Arrays ---
            int[] numbers = { 1, 2, 3, 4, 5 };
            int[,] matrix = { { 1, 2 }, { 3, 4 } };
            
            // --- Lists and Collections ---
            List<string> fruits = new List<string> { "apple", "banana", "orange" };
            fruits.Add("grape");
            fruits.Remove("banana");
            Console.WriteLine($"Fruit count: {fruits.Count}");
            
            Dictionary<string, int> ages = new Dictionary<string, int>
            {
                {"Alice", 30},
                {"Bob", 25}
            };
            ages["Charlie"] = 35;

            // --- Control Flow ---
            if (age >= 18)
                Console.WriteLine("Adult");
            else
                Console.WriteLine("Minor");

            string category = age < 13 ? "Child" : age < 20 ? "Teen" : "Adult";
            Console.WriteLine(category);

            switch (grade)
            {
                case 'A':
                    Console.WriteLine("Excellent");
                    break;
                case 'B':
                    Console.WriteLine("Good");
                    break;
                default:
                    Console.WriteLine("OK");
                    break;
            }

            // Loops
            for (int i = 0; i < 3; i++)
                Console.Write(i + " ");
            Console.WriteLine();

            foreach (string fruit in fruits)
                Console.Write(fruit + " ");

            // --- Functions and Lambdas ---
            Func<int, int, int> add = (a, b) => a + b;
            Console.WriteLine($"Add(2,3) = {add(2, 3)}");

            Action<string> print = Console.WriteLine;
            print("Hello from lambda.");

            // Local function
            int Multiply(int a, int b) => a * b;
            Console.WriteLine(Multiply(4, 5));

            // --- Classes and Objects ---
            Dog myDog = new Dog("Rex", "German Shepherd");
            Console.WriteLine(myDog.Speak());
            Console.WriteLine(Dog.Info());

            Point p1 = new Point(3, 4);
            Point p2 = new Point(1, 2);
            Point sum = p1 + p2;
            Console.WriteLine(sum);

            Box<int> intBox = new Box<int>(42);
            Console.WriteLine(intBox);

            // --- Interfaces ---
            IAnimal animal = myDog; // polymorphism
            Console.WriteLine(animal.Speak());

            // --- Delegates and Events ---
            Publisher pub = new Publisher();
            pub.OnNotify += (msg) => Console.WriteLine($"Event received: {msg}");
            pub.RaiseEvent("Test message");

            // --- LINQ ---
            var longFruits = fruits.Where(f => f.Length > 5).Select(f => f.ToUpper());
            Console.WriteLine("Long fruits: " + string.Join(", ", longFruits));

            var sorted = fruits.OrderBy(f => f.Length);
            Console.WriteLine("Sorted by length: " + string.Join(", ", sorted));

            var fruitGroups = fruits.GroupBy(f => f[0]);
            foreach (var group in fruitGroups)
                Console.WriteLine($"Fruits starting with {group.Key}: {string.Join(", ", group)}");

            // --- Nullable Types ---
            int? maybeNull = null;
            maybeNull ??= 10;
            Console.WriteLine(maybeNull);

            // --- Error Handling ---
            try
            {
                throw new InvalidOperationException("Something went wrong");
            }
            catch (InvalidOperationException ex)
            {
                Console.WriteLine($"Caught: {ex.Message}");
            }
            finally
            {
                Console.WriteLine("Finally block executed.");
            }

            // --- File I/O ---
            string path = "test.txt";
            File.WriteAllText(path, "Hello, file system!");
            string content = File.ReadAllText(path);
            Console.WriteLine($"File content: {content}");

            // --- Regular Expressions ---
            string email = "user@example.com";
            Regex emailRegex = new Regex(@"^[^\s@]+@[^\s@]+\.[^\s@]+$");
            Console.WriteLine($"Email valid: {emailRegex.IsMatch(email)}");

            // --- Async/Await ---
            DataFetcher fetcher = new DataFetcher();
            string fetched = await fetcher.FetchDataAsync("http://api.example.com");
            Console.WriteLine(fetched);

            // --- Parallel Loop ---
            Parallel.For(0, 5, i => Console.WriteLine("Parallel " + i));

            // --- Simple Algorithms ---
            int[] arr = { 4, 2, 8, 5, 1 };
            Array.Sort(arr);
            Console.WriteLine("Sorted: " + string.Join(", ", arr));
            int max = arr.Max();
            Console.WriteLine("Max: " + max);

            // Factorial
            static int Factorial(int n) => n <= 1 ? 1 : n * Factorial(n - 1);
            Console.WriteLine($"Factorial(5) = {Factorial(5)}");

            // Fibonacci
            static int Fibonacci(int n)
            {
                if (n <= 1) return n;
                int a = 0, b = 1, temp;
                for (int i = 2; i <= n; i++)
                {
                    temp = a + b;
                    a = b;
                    b = temp;
                }
                return b;
            }
            Console.WriteLine($"Fibonacci(7) = {Fibonacci(7)}");

            // Palindrome
            static bool IsPalindrome(string str)
            {
                string cleaned = Regex.Replace(str.ToLower(), "[^a-z0-9]", "");
                return cleaned == new string(cleaned.Reverse().ToArray());
            }
            Console.WriteLine(IsPalindrome("Racecar")); // true

            // --- Final Output ---
            Console.WriteLine("C# training sample complete.");
        }
    }
}
