            <h2>Web Components (Custom Elements)</h2>

            <!-- A custom element defined in the script below -->
            <user-card name="Alice" avatar="alice.webp" email="alice@example.com"></user-card>

            <template id="user-card-template">
                <style>
                    .card { border: 1px solid #ccc; padding: 1rem; display: inline-flex; gap: 1rem; align-items: center; border-radius: 8px; }
                    img { width: 48px; height: 48px; border-radius: 50%; }
                </style>
                <div class="card">
                    <img alt="" src="">
                    <div>
                        <strong></strong>
                        <small></small>
                    </div>
                </div>
            </template>

            <script>
                class UserCard extends HTMLElement {
                    constructor() {
                        super();
                        const template = document.getElementById('user-card-template').content.cloneNode(true);
                        this.attachShadow({ mode: 'open' }).appendChild(template);
                    }
                    connectedCallback() {
                        this.shadowRoot.querySelector('img').src = this.getAttribute('avatar') || '';
                        this.shadowRoot.querySelector('strong').textContent = this.getAttribute('name') || '';
                        this.shadowRoot.querySelector('small').textContent = this.getAttribute('email') || '';
                    }
                }
                customElements.define('user-card', UserCard);
            </script>
        </section>
